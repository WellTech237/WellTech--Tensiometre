import 'package:fl_chart/fl_chart.dart';
import 'package:flutter/material.dart';

import '../../core/classification.dart';
import '../../data/models.dart';
import '../../services/app_state.dart';
import '../../theme.dart';
import '../widgets.dart';

enum Serie { tas, tad, pouls }

class _Jour {
  final DateTime date;
  final List<Mesure> mesures; // hors mesures de crise
  final bool crise;
  final List<String> symptomes;
  _Jour(this.date, this.mesures, this.crise, this.symptomes);

  int? moy(int Function(Mesure) f) => mesures.isEmpty ? null : (mesures.map(f).reduce((a, b) => a + b) / mesures.length).round();
}

/// Écran 10 : graphiques détaillés (semaine / mois, TAS / TAD / pouls,
/// détail d'un jour, statistiques, matin / soir, historique).
class GraphiquesScreen extends StatefulWidget {
  const GraphiquesScreen({super.key});
  @override
  State<GraphiquesScreen> createState() => _GraphiquesScreenState();
}

class _GraphiquesScreenState extends State<GraphiquesScreen> {
  bool _mois = false;
  Serie? _serie = Serie.tad;
  int? _jourChoisi;

  static const _couleurs = {Serie.tas: Wt.violet, Serie.tad: Wt.tad, Serie.pouls: Wt.orange};
  static const _noms = {Serie.tas: 'TAS', Serie.tad: 'TAD', Serie.pouls: 'Pouls'};

  Future<List<Mesure>> _charger() {
    final n = _mois ? 30 : 7;
    final now = DateTime.now();
    return AppState.instance.db.mesuresDepuis(DateTime(now.year, now.month, now.day).subtract(Duration(days: n - 1)));
  }

  List<_Jour> _parJour(List<Mesure> toutes) {
    final n = _mois ? 30 : 7;
    final now = DateTime.now();
    final debut = DateTime(now.year, now.month, now.day).subtract(Duration(days: n - 1));
    return List.generate(n, (i) {
      final d = debut.add(Duration(days: i));
      final duJour = toutes.where((m) => m.date.year == d.year && m.date.month == d.month && m.date.day == d.day).toList();
      return _Jour(
        d,
        duJour.where((m) => m.episodeId == null).toList(),
        duJour.any((m) => m.episodeId != null),
        duJour.expand((m) => m.symptomes).toSet().toList(),
      );
    });
  }

  int? _valeur(_Jour j, Serie s) => switch (s) {
        Serie.tas => j.moy((m) => m.tas),
        Serie.tad => j.moy((m) => m.tad),
        Serie.pouls => j.moy((m) => m.pouls),
      };

  @override
  Widget build(BuildContext context) {
    return FutureBuilder<List<Mesure>>(
      future: _charger(),
      builder: (context, snap) {
        final toutes = snap.data ?? const <Mesure>[];
        final jours = _parJour(toutes);
        final avecDonnees = jours.where((j) => j.mesures.isNotEmpty).toList();
        final choisi = (_jourChoisi != null && _jourChoisi! < jours.length) ? _jourChoisi! : (avecDonnees.isEmpty ? jours.length - 1 : jours.indexOf(avecDonnees.last));
        return ListView(padding: const EdgeInsets.fromLTRB(20, 16, 20, 24), children: [
          Row(mainAxisAlignment: MainAxisAlignment.spaceBetween, children: [
            const Text('Graphiques', style: TextStyle(fontSize: 26, fontWeight: FontWeight.w800)),
            BoutonEcouter(texte: () => _resume(jours)),
          ]),
          const SizedBox(height: 12),
          SegmentedButton<bool>(
            segments: const [ButtonSegment(value: false, label: Text('Semaine')), ButtonSegment(value: true, label: Text('Mois'))],
            selected: {_mois},
            onSelectionChanged: (s) => setState(() {
              _mois = s.first;
              _jourChoisi = null;
            }),
          ),
          const SizedBox(height: 12),
          if (avecDonnees.isEmpty)
            const Carte(child: Text('Pas encore de mesure sur cette période. Vos graphiques apparaîtront après vos premières mesures.', style: TextStyle(height: 1.5)))
          else ...[
            Carte(
              padding: const EdgeInsets.fromLTRB(8, 12, 12, 8),
              child: Column(children: [
                Row(children: [
                  for (final s in Serie.values) ...[
                    Expanded(child: _boutonSerie(s)),
                    if (s != Serie.pouls) const SizedBox(width: 6),
                  ],
                ]),
                const SizedBox(height: 10),
                _reperes(jours),
                SizedBox(height: 230, child: _graphique(jours, choisi)),
                const SizedBox(height: 6),
                const Text('Touchez un point pour voir le détail du jour · C crise · S symptôme', style: TextStyle(fontSize: 11, color: Wt.gris, fontWeight: FontWeight.w600)),
              ]),
            ),
            const SizedBox(height: 12),
            _detailJour(jours[choisi]),
            const SizedBox(height: 12),
            _stats(jours),
            const SizedBox(height: 12),
            _matinSoir(toutes),
            const SizedBox(height: 14),
            const Etiquette('Historique'),
            const SizedBox(height: 8),
            Carte(
              padding: const EdgeInsets.symmetric(horizontal: 4, vertical: 4),
              child: Column(children: [
                for (final m in toutes.reversed.take(10))
                  ListTile(
                    leading: Icon(Icons.circle, size: 12, color: couleursGrade(Grade.values.byName(m.grade)).plein),
                    title: Text('${m.tas}/${m.tad} · pouls ${m.pouls}${m.episodeId != null ? ' · crise' : ''}', style: const TextStyle(fontWeight: FontWeight.w800)),
                    subtitle: Text('${formatDate(m.date)}${m.symptomes.isEmpty ? '' : ' · ${m.symptomes.join(', ')}'}'),
                    trailing: PuceGrade(Grade.values.byName(m.grade)),
                  ),
              ]),
            ),
          ],
        ]);
      },
    );
  }

  Widget _boutonSerie(Serie s) {
    final actif = _serie == s;
    return OutlinedButton(
      style: OutlinedButton.styleFrom(
        minimumSize: const Size(0, 44),
        backgroundColor: actif ? Wt.violetClair : Wt.blanc,
        side: BorderSide(color: actif ? _couleurs[s]! : Wt.bordure, width: 2),
        padding: const EdgeInsets.symmetric(horizontal: 6),
      ),
      onPressed: () => setState(() => _serie = actif ? null : s),
      child: Row(mainAxisSize: MainAxisSize.min, children: [
        Container(width: 14, height: 4, color: _couleurs[s]),
        const SizedBox(width: 6),
        Text(_noms[s]!),
      ]),
    );
  }

  /// Repères C (crise) et S (symptôme) au-dessus du graphique.
  Widget _reperes(List<_Jour> jours) => Padding(
        padding: const EdgeInsets.only(left: 34),
        child: Row(children: [
          for (final j in jours)
            Expanded(
              child: Center(
                child: j.crise
                    ? _pastille('C', Wt.rouge, Wt.blanc)
                    : (j.symptomes.isNotEmpty ? _pastille('S', Wt.orangeClair, Wt.orangeTexte) : const SizedBox(height: 18)),
              ),
            ),
        ]),
      );

  Widget _pastille(String t, Color fond, Color encre) => Container(
        width: 18,
        height: 18,
        alignment: Alignment.center,
        decoration: BoxDecoration(color: fond, shape: BoxShape.circle),
        child: Text(t, style: TextStyle(color: encre, fontSize: 10, fontWeight: FontWeight.w800)),
      );

  /// Bandes de couleur selon la courbe choisie (TAS et TAD n'ont pas les mêmes seuils).
  List<HorizontalRangeAnnotation> _bandes() {
    final s = _serie ?? Serie.tas;
    final lim = switch (s) {
      Serie.tas => [(180.0, 200.0, Wt.rougeClair), (140.0, 180.0, Wt.orangeClair), (130.0, 140.0, Wt.jaune), (40.0, 130.0, Wt.vertClair)],
      Serie.tad => [(120.0, 200.0, Wt.rougeClair), (90.0, 120.0, Wt.orangeClair), (80.0, 90.0, Wt.jaune), (40.0, 80.0, Wt.vertClair)],
      Serie.pouls => [(100.0, 200.0, Wt.orangeClair), (60.0, 100.0, Wt.vertClair), (40.0, 60.0, Wt.jaune)],
    };
    return [for (final l in lim) HorizontalRangeAnnotation(y1: l.$1, y2: l.$2, color: l.$3)];
  }

  Widget _graphique(List<_Jour> jours, int choisi) {
    LineChartBarData barre(Serie s) {
      final attenue = _serie != null && _serie != s;
      final c = _couleurs[s]!;
      return LineChartBarData(
        spots: [
          for (var i = 0; i < jours.length; i++)
            if (_valeur(jours[i], s) != null) FlSpot(i.toDouble(), _valeur(jours[i], s)!.toDouble()) else FlSpot.nullSpot,
        ],
        color: attenue ? c.withAlpha(40) : c,
        barWidth: s == Serie.pouls ? 2 : 3,
        dashArray: s == Serie.pouls ? [3, 5] : null,
        dotData: FlDotData(show: !_mois || !attenue),
      );
    }

    return LineChart(LineChartData(
      minX: 0,
      maxX: (jours.length - 1).toDouble(),
      minY: 40,
      maxY: 200,
      rangeAnnotations: RangeAnnotations(horizontalRangeAnnotations: _bandes()),
      gridData: const FlGridData(show: false),
      borderData: FlBorderData(show: false),
      extraLinesData: ExtraLinesData(verticalLines: [VerticalLine(x: choisi.toDouble(), color: Wt.encre, strokeWidth: 1.5, dashArray: [4, 4])]),
      titlesData: FlTitlesData(
        topTitles: const AxisTitles(sideTitles: SideTitles(showTitles: false)),
        rightTitles: const AxisTitles(sideTitles: SideTitles(showTitles: false)),
        leftTitles: AxisTitles(
          sideTitles: SideTitles(showTitles: true, reservedSize: 34, interval: 40, getTitlesWidget: (v, meta) => SideTitleWidget(axisSide: meta.axisSide, child: Text(v.toInt().toString(), style: const TextStyle(fontSize: 10, color: Wt.gris)))),
        ),
        bottomTitles: AxisTitles(
          sideTitles: SideTitles(
            showTitles: true,
            interval: 1,
            reservedSize: 22,
            getTitlesWidget: (v, meta) {
              final i = v.toInt();
              if (i < 0 || i >= jours.length) return const SizedBox.shrink();
              final d = jours[i].date;
              final txt = _mois ? (i % 7 == 0 ? '${d.day}/${d.month}' : '') : const ['L', 'M', 'M', 'J', 'V', 'S', 'D'][d.weekday - 1];
              return SideTitleWidget(axisSide: meta.axisSide, child: Text(txt, style: TextStyle(fontSize: 10, fontWeight: i == choisi ? FontWeight.w800 : FontWeight.w600, color: i == choisi ? Wt.encre : Wt.gris)));
            },
          ),
        ),
      ),
      lineTouchData: LineTouchData(
        touchCallback: (event, reponse) {
          final spots = reponse?.lineBarSpots;
          if (event is FlTapUpEvent && spots != null && spots.isNotEmpty) {
            setState(() => _jourChoisi = spots.first.x.toInt());
          }
        },
        touchTooltipData: LineTouchTooltipData(
          getTooltipColor: (_) => Wt.encre,
          getTooltipItems: (spots) => spots.map((s) => LineTooltipItem(s.y.toInt().toString(), const TextStyle(color: Wt.blanc, fontWeight: FontWeight.w800))).toList(),
        ),
      ),
      lineBarsData: [barre(Serie.tas), barre(Serie.tad), barre(Serie.pouls)],
    ));
  }

  Widget _detailJour(_Jour j) {
    final tas = j.moy((m) => m.tas), tad = j.moy((m) => m.tad), pouls = j.moy((m) => m.pouls);
    final g = (tas != null && tad != null) ? classerPour(tas, tad, enceinte: AppState.instance.profil.grossesseActive) : null;
    const jours = ['Lundi', 'Mardi', 'Mercredi', 'Jeudi', 'Vendredi', 'Samedi', 'Dimanche'];
    return Carte(
      couleur: Wt.encre,
      child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
        Row(mainAxisAlignment: MainAxisAlignment.spaceBetween, children: [
          Text('${jours[j.date.weekday - 1]} ${j.date.day}/${j.date.month}', style: const TextStyle(color: Wt.blanc, fontWeight: FontWeight.w800)),
          if (g != null) PuceGrade(g),
        ]),
        const SizedBox(height: 10),
        if (tas == null)
          const Text('Pas de mesure ce jour-là.', style: TextStyle(color: Color(0xFFDDE2EA)))
        else ...[
          Row(children: [
            _valeurDetail('TAS', '$tas'),
            _valeurDetail('TAD', '$tad'),
            _valeurDetail('Pouls', '$pouls'),
            _valeurDetail('Différ.', '${tas - tad!}'),
          ]),
          const SizedBox(height: 8),
          Text('Symptômes : ${j.symptomes.isEmpty ? 'aucun' : j.symptomes.join(', ')}\nMesures du jour : ${j.mesures.length}${j.crise ? ' + épisode de crise' : ''}',
              style: const TextStyle(color: Color(0xFFDDE2EA), height: 1.5, fontSize: 13)),
        ],
      ]),
    );
  }

  Widget _valeurDetail(String k, String v) => Expanded(
        child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
          Text(k, style: const TextStyle(color: Color(0xFFB7BFCC), fontSize: 11, fontWeight: FontWeight.w700)),
          Text(v, style: const TextStyle(color: Wt.blanc, fontSize: 18, fontWeight: FontWeight.w800)),
        ]),
      );

  Widget _stats(List<_Jour> jours) {
    final s = _serie ?? Serie.tas;
    final vals = jours.map((j) => _valeur(j, s)).whereType<int>().toList();
    final seuil = switch (s) { Serie.tas => 140, Serie.tad => 90, Serie.pouls => 100 };
    final moy = vals.isEmpty ? 0 : (vals.reduce((a, b) => a + b) / vals.length).round();
    return Carte(
      child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
        Etiquette('${_noms[s]} sur ${_mois ? '30' : '7'} jours'),
        const SizedBox(height: 8),
        Row(children: [
          _stat('Min', vals.isEmpty ? '—' : '${vals.reduce((a, b) => a < b ? a : b)}'),
          _stat('Max', vals.isEmpty ? '—' : '${vals.reduce((a, b) => a > b ? a : b)}'),
          _stat('Moyenne', vals.isEmpty ? '—' : '$moy'),
          _stat('≥ $seuil', '${vals.where((v) => v >= seuil).length}/${vals.length}'),
        ]),
      ]),
    );
  }

  Widget _stat(String k, String v) => Expanded(
        child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
          Text(k, style: const TextStyle(fontSize: 11, color: Wt.gris, fontWeight: FontWeight.w700)),
          Text(v, style: const TextStyle(fontSize: 17, fontWeight: FontWeight.w800)),
        ]),
      );

  Widget _matinSoir(List<Mesure> toutes) {
    final normales = toutes.where((m) => m.episodeId == null);
    String moy(Iterable<Mesure> l) {
      if (l.isEmpty) return '—';
      final t = (l.map((m) => m.tas).reduce((a, b) => a + b) / l.length).round();
      final d = (l.map((m) => m.tad).reduce((a, b) => a + b) / l.length).round();
      return '$t/$d';
    }

    return Row(children: [
      Expanded(child: Carte(couleur: const Color(0xFFFFF8E6), child: _blocMoyenne('Matin (moy.)', moy(normales.where((m) => m.date.hour < 12)), Wt.jauneTexte))),
      const SizedBox(width: 10),
      Expanded(child: Carte(couleur: const Color(0xFFEEF0FB), child: _blocMoyenne('Soir (moy.)', moy(normales.where((m) => m.date.hour >= 12)), Wt.violetFonce))),
    ]);
  }

  Widget _blocMoyenne(String titre, String v, Color c) => Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
        Etiquette(titre, couleur: c),
        const SizedBox(height: 4),
        Text(v, style: const TextStyle(fontSize: 22, fontWeight: FontWeight.w800)),
      ]);

  String _resume(List<_Jour> jours) {
    final tas = jours.map((j) => j.moy((m) => m.tas)).whereType<int>().toList();
    final tad = jours.map((j) => j.moy((m) => m.tad)).whereType<int>().toList();
    if (tas.isEmpty) return 'Pas encore de mesure sur cette période.';
    final mt = (tas.reduce((a, b) => a + b) / tas.length).round();
    final md = (tad.reduce((a, b) => a + b) / tad.length).round();
    final crises = jours.where((j) => j.crise).length;
    return 'Sur les ${_mois ? '30' : '7'} derniers jours, votre tension moyenne est de $mt sur $md. '
        'Vous avez mesuré ${tas.length} jour(s). ${crises == 0 ? 'Aucune crise.' : '$crises jour(s) avec une crise.'}';
  }
}
