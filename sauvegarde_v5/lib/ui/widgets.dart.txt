class Carte extends StatelessWidget {
  final Widget child;
  final Color couleur;
  final EdgeInsetsGeometry padding;
  final VoidCallback? onTap;
  const Carte({super.key, required this.child, this.couleur = Wt.blanc, this.padding = const EdgeInsets.all(16), this.onTap});

  @override
  Widget build(BuildContext context) {
    final c = Container(
      width: double.infinity,
      padding: padding,
      decoration: BoxDecoration(color: couleur, borderRadius: BorderRadius.circular(24)),
      child: child,
    );
    if (onTap == null) return c;
    return Material(color: Colors.transparent, child: InkWell(borderRadius: BorderRadius.circular(24), onTap: onTap, child: c));
  }
}

/// Bouton « Écouter » : lit le texte à voix haute.
class BoutonEcouter extends StatelessWidget {
  final String Function() texte;
  final bool compact;
  final Color fond;
  final Color encre;
  const BoutonEcouter({super.key, required this.texte, this.compact = false, this.fond = Wt.blanc, this.encre = Wt.encre});

  @override
  Widget build(BuildContext context) {
    final icone = Icon(Icons.volume_up_outlined, color: encre == Wt.encre ? Wt.violet : encre, size: 22);
    if (compact) {
      return Semantics(
        label: 'Écouter',
        button: true,
        child: Material(
          color: fond,
          shape: const CircleBorder(),
          child: InkWell(customBorder: const CircleBorder(), onTap: () => Voix.instance.lire(texte()), child: SizedBox(width: 48, height: 48, child: icone)),
        ),
      );
    }
    return TextButton.icon(
      onPressed: () => Voix.instance.lire(texte()),
      icon: icone,
      label: Text('Écouter', style: TextStyle(fontWeight: FontWeight.w800, color: encre)),
      style: TextButton.styleFrom(backgroundColor: fond, minimumSize: const Size(0, 44), shape: const StadiumBorder(), padding: const EdgeInsets.symmetric(horizontal: 14)),
    );
  }
}

class CouleursGrade {
  final Color fond, texte, plein;
  const CouleursGrade(this.fond, this.texte, this.plein);
}

CouleursGrade couleursGrade(Grade g) => switch (g) {
      Grade.normale => const CouleursGrade(Wt.vertClair, Wt.vertTexte, Wt.vert),
      Grade.elevee => const CouleursGrade(Color(0xFFF3F4D6), Color(0xFF4F5A00), Color(0xFF8A9A00)),
      Grade.grade1 => const CouleursGrade(Wt.jaune, Wt.jauneTexte, Color(0xFFD99A00)),
      Grade.grade2 => const CouleursGrade(Wt.orangeClair, Wt.orangeTexte, Wt.orange),
      Grade.grade3 => const CouleursGrade(Wt.rougeClair, Wt.rougeTexte, Wt.rouge),
    };

Color couleurNiveau(int n) => const [Wt.vert, Color(0xFF8A6100), Wt.orange, Wt.rouge][(n - 1).clamp(0, 3)];

class PuceGrade extends StatelessWidget {
  final Grade grade;
  final bool long;
  const PuceGrade(this.grade, {super.key, this.long = false});

  @override
  Widget build(BuildContext context) {
    final c = couleursGrade(grade);
    return Container(
      padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 6),
      decoration: BoxDecoration(color: c.fond, borderRadius: BorderRadius.circular(999)),
      child: Text(long ? grade.libelle : grade.court, style: TextStyle(color: c.texte, fontWeight: FontWeight.w800, fontSize: 13)),
    );
  }
}

/// Puce à cocher (symptômes, maladies).
class Puce extends StatelessWidget {
  final String texte;
  final bool active;
  final VoidCallback onTap;
  final Color couleur;
  const Puce({super.key, required this.texte, required this.active, required this.onTap, this.couleur = Wt.violet});

  @override
  Widget build(BuildContext context) => Semantics(
        selected: active,
        button: true,
        child: Material(
          color: active ? couleur.withAlpha(28) : Wt.blanc,
          shape: StadiumBorder(side: BorderSide(color: active ? couleur : Wt.bordure, width: 2)),
          child: InkWell(
            customBorder: const StadiumBorder(),
            onTap: onTap,
            child: ConstrainedBox(
              constraints: const BoxConstraints(minHeight: 44),
              child: Padding(
                padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 10),
                child: Text(texte, style: TextStyle(fontWeight: FontWeight.w700, color: active ? couleur : Wt.encre)),
              ),
            ),
          ),
        ),
      );
}

/// Pastille réseau « En ligne / Hors ligne ».

  @override
  Widget build(BuildContext context) => Container(
        padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 8),
        decoration: BoxDecoration(color: enLigne ? Wt.vertClair : Wt.jaune, borderRadius: BorderRadius.circular(999)),
        child: Row(mainAxisSize: MainAxisSize.min, children: [
          Container(width: 8, height: 8, decoration: BoxDecoration(color: enLigne ? Wt.vert : Wt.orange, shape: BoxShape.circle)),
          const SizedBox(width: 6),
          Text(enLigne ? 'En ligne' : 'Hors ligne', style: TextStyle(fontSize: 12, fontWeight: FontWeight.w800, color: enLigne ? Wt.vertTexte : Wt.jauneTexte)),
        ]),
      );
}

String formatDate(DateTime d) {
  const jours = ['lun.', 'mar.', 'mer.', 'jeu.', 'ven.', 'sam.', 'dim.'];
  return '${jours[d.weekday - 1]} ${d.day}/${d.month} · ${d.hour} h ${d.minute.toString().padLeft(2, '0')}';
}
